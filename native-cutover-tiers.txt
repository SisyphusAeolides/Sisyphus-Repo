# Staged consumer rebuild DAG for native RustD cutover.
# Packages are ordered by dependency foundations. Use with:
#   python3 scripts/audit-systemd-closure.py --manifest native-cutover-packages.txt
#   bash scripts/check-consumer-tiers.sh --tier T2
#   bash scripts/check-consumer-tiers.sh --tier T3 --fail-on-consumers
#
# T0 lands with rustd-libs/rustd-devel. Later tiers rebuild against those APIs
# (or disable optional journal/udev plugins) before systemd-libs can leave.
